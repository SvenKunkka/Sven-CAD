cmake_minimum_required(VERSION 3.24)
project(PersonalCADQtWidgets VERSION 6.11.2 LANGUAGES C CXX OBJCXX)

if(NOT APPLE OR NOT CMAKE_OSX_ARCHITECTURES STREQUAL "arm64")
    message(FATAL_ERROR "The personal CAD Qt Widgets comparison is for macOS arm64.")
endif()
if(NOT IS_DIRECTORY "${QTBASE_SOURCE}/src/widgets")
    message(FATAL_ERROR "Set QTBASE_SOURCE to the verified Qt 6.11.2 C source tree.")
endif()

set(QT_SKIP_AUTO_PLUGIN_INCLUSION ON)
find_package(Qt6 6.11.2 EXACT REQUIRED COMPONENTS Core CorePrivate Gui GuiPrivate)
find_package(Qt6WidgetsTools 6.11.2 EXACT REQUIRED)
find_package(Qt6BuildInternals 6.11.2 EXACT REQUIRED)
qt_policy(SET QTP0003 NEW)
qt_internal_setup_default_target_function_options()

set(QT_BUILD_DIR "${CMAKE_BINARY_DIR}")
set(QT_CMAKE_DIR "${QTBASE_SOURCE}/cmake")
set(QT_REPO_MODULE_VERSION "${PROJECT_VERSION}")
set(CMAKE_AUTOMOC ON)
set(CMAKE_MACOSX_RPATH ON)
set(CMAKE_INSTALL_NAME_DIR "@rpath")
set(CMAKE_BUILD_WITH_INSTALL_RPATH ON)
set(CMAKE_INSTALL_RPATH "@loader_path/")
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/lib")
set(FEATURE_framework OFF)

add_custom_target(sync_headers)
add_subdirectory("${QTBASE_SOURCE}/src/widgets" upstream-widgets)
